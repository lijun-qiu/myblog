## 项目概述

这是一个轻量级的前端性能监控SDK，专门用于监控Web应用的稳定性和错误情况。该SDK采用模块化设计，能够自动捕获JavaScript错误、网络请求异常、白屏问题等关键监控指标，为开发者提供全面的前端应用健康状态监控。

### �� 监控目标
- **稳定性监控**：JS错误、资源加载错误、Promise错误
- **网络监控**：XHR请求异常、响应时间统计
- **用户体验监控**：白屏检测、页面可用性
- **错误定位**：提供详细的错误上下文信息

## 核心模块详解

### 1. JavaScript错误监控 (`jsError.js`)

#### 功能概述
负责捕获和处理JavaScript运行时错误，包括同步错误、异步错误和资源加载错误。

#### 核心实现

**全局错误监听**
```javascript
window.addEventListener('error', function (event){
    let lastEvent = getLastEvent(); // 捕获最后一个交互事件
    if( event.target && (event.target.src || event.target.href)) {
        // 资源加载错误处理
        let errorLog = {
            kind: 'stability',
            type: 'error',
            errorType: 'resourceError',
            filename: event.target.src || event.target.href,
            tagName: event.target.tagName,
            selector: getSelector(event.target)
        }
        tracker.send(errorLog);
    }else{
        // JS执行错误处理
        let errorLog = {
            kind: 'stability',
            type: 'error',
            errorType: 'jsError',
            message: event.message,
            filename: event.filename,
            position: `${event.lineno}:${event.colno}`,
            stack: event.error && getLines(event.error.stack),
            selector: lastEvent ? getSelector(lastEvent.path) : ""
        }
        tracker.send(errorLog);
    }
}, true);
```

**Promise错误捕获**
```javascript
window.addEventListener('unhandledrejection', (event) => {
    let lastEvent = getLastEvent();
    let message;
    let reason = event.reason;
    let lineno = 0;
    let colno = 0;
    let filename;
    let stack;
    
    if(typeof event.reason === 'string') {
        message = reason;
    }else if(typeof reason === 'object'){
        if(reason.stack){
            let matchResult = reason.stack.match(/at\s+(.+):(\d+):(\d+)/);
            filename = matchResult[1];
            lineno = matchResult[2];
            colno = matchResult[3];
        }
        message = reason.message;
        stack = getLines(reason.stack);
    }
    
    let errorLog = {
        kind: 'stability',
        type: 'error',
        errorType: 'promiseError',
        message: message,
        filename: filename,
        position: `${lineno}:${colno}`,
        stack: stack,
        selector: lastEvent ? getSelector(lastEvent.path) : ""
    }
    tracker.send(errorLog)
}, true)
```

**堆栈信息处理**
```javascript
function getLines(stack) {
    return stack.split('\n').slice(1).map(item=>item.replace(/^\s+at\s+/g, "")).join('^');
}
```

#### 监控数据类型
- **JS执行错误** (`errorType: 'jsError'`)
  - 错误消息 (`message`)
  - 文件名 (`filename`)
  - 错误位置 (`position: "行:列"`)
  - 堆栈信息 (`stack`)
  - 最后操作元素 (`selector`)

- **资源加载错误** (`errorType: 'resourceError'`)
  - 资源URL (`filename`)
  - 标签名 (`tagName`)
  - 元素选择器 (`selector`)

- **Promise错误** (`errorType: 'promiseError'`)
  - 错误消息和堆栈信息
  - 文件名和位置信息

#### 技术亮点
- **错误分类**：自动区分不同类型的错误
- **堆栈解析**：提取并格式化错误堆栈信息
- **上下文捕获**：记录用户最后操作的元素
- **正则匹配**：通过正则表达式解析Promise错误信息

### 2. 网络请求监控 (`xhr.js`)

#### 功能概述
通过拦截XMLHttpRequest来监控网络请求的状态、性能和错误情况。

#### 核心实现

**请求拦截**
```javascript
let XMLHttpRequest = window.XMLHttpRequest;
let oldOpen = XMLHttpRequest.prototype.open;
XMLHttpRequest.prototype.open = function (method, url, async) {
    // 把上报接口过滤掉，避免循环上报
    if (!url.match(/plaque_log/)) {
        this.logData = { method, url, async };
    }
    return oldOpen.apply(this, arguments);
};
```

**响应监控**
```javascript
let oldSend = XMLHttpRequest.prototype.send;
XMLHttpRequest.prototype.send = function (body) {
    if (this.logData) {
        let startTime = Date.now(); // 发送前记录开始时间
        let handler = (type) => (event) => {
            // 持续时间
            let duration = Date.now() - startTime;
            let status = this.status;  // 200 500
            let statusText = this.statusText;  // Ok Server Error
            
            // 接口请求监控
            tracker.send({
                kind: "stability",
                type: "xhr",
                eventType: type,  // load, error, abort 请求类型
                pathname: this.logData.url,
                status: status + "-" + statusText,
                duration,
                response: this.response ? JSON.stringify(this.response) : "",
                params: body || "",
            });
        };
        
        // 监听成功，失败，的所有接口，这里只监控error 
        this.addEventListener("error", handler("error"), false);
    }
    return oldSend.apply(this, arguments);
};
```

#### 监控数据类型
- **请求信息**：方法、URL、异步标志
- **响应信息**：状态码、状态文本、响应内容
- **性能指标**：请求持续时间
- **错误类型**：error事件

#### 技术亮点
- **原型链拦截**：通过重写XMLHttpRequest原型方法实现监控
- **循环上报防护**：过滤监控接口本身，避免无限循环
- **性能统计**：精确计算请求耗时
- **选择性监控**：只监控error事件，减少数据量
- **时间记录**：在send方法中记录开始时间

### 3. 白屏检测 (`blankScreen.js`)

#### 功能概述
通过采样检测页面关键位置来判断是否存在白屏问题。

#### 核心实现

**选择器生成**
```javascript
function getSelector(element) {
    const {id, className, nodeName} = element;
    if (id) {
        return "#" + id;
    } else if (className) {
        // 过滤空白符 + 拼接 可能多个className  a  b   c => .a.b.c
        return (
            ("." + className )
                .split(" ")
                .filter((item) => !!item)
                .join(".")
        );
    } else {
        return nodeName.toLowerCase();
    }
}
```

**采样检测算法**
```javascript
function isWrapper(element) {
    let selector = getSelector(element);
    if (wrapperElements.indexOf(selector) !== -1) {
        emptyPoints++;
    }
}

// 刚开始页面内容为空，监听接口加载，等页面渲染完成，再去做判断
onload(function () {
    let xElements, yElements;
    for (let i = 0; i < 9; i++) {
        xElements = document.elementsFromPoint(
            (window.innerWidth * i) / 10,
            window.innerHeight / 2
        );
        yElements = document.elementsFromPoint(
            window.innerWidth / 2,
            (window.innerHeight * i) / 10
        );
        isWrapper(xElements[0]);
        isWrapper(yElements[0]);
    }
    
    // 白屏 总共18个点  如果空白点数量大于 16个 代表白屏
    if (emptyPoints >= 16) {
        const centerElements = document.elementsFromPoint(
            window.innerWidth / 2,
            window.innerHeight / 2
        );

        // 白屏监控
        tracker.send({
            kind: "stability",
            type: "blank",
            emptyPoints: emptyPoints + "",
            screen: window.screen.width + "X" + window.screen.height,
            viewPoint: window.innerWidth + "X" + window.innerHeight,
            selector: getSelector(centerElements[0]),
        });
    }
});
```

#### 监控数据类型
- **空白点数量** (`emptyPoints`)
- **屏幕分辨率** (`screen`)
- **视口大小** (`viewPoint`)
- **中心点选择器** (`selector`)

#### 技术亮点
- **十字型采样**：在页面关键位置进行采样检测
- **延迟检测**：等待页面完全加载后再检测
- **阈值判断**：通过空白点数量阈值判断白屏
- **环境信息**：记录屏幕和视口信息
- **elementsFromPoint API**：使用现代API获取指定位置元素

### 2. DOM选择器生成 (`getSelector.js`)

#### 功能概述
生成DOM元素的完整选择器路径，用于错误定位。

#### 核心实现

**路径构建算法**
```javascript
function getSelectors(path) {
    // 反转 + 过滤 + 映射 + 拼接
    return path
        .reverse()
        .filter((element) => {
            return element !== document && element !== window;
        })
        .map((element) => {
            console.log("element", element.nodeName);
            let selector = "";
            if (element.id) {
                return `${element.nodeName.toLowerCase()}#${element.id}`;
            } else if (element.className && typeof element.className === "string") {
                return `${element.nodeName.toLowerCase()}.${element.className}`;
            } else {
                selector = element.nodeName.toLowerCase();
            }
            return selector;
        })
        .join(" ");
}

export default function (pathsOrTarget) {
    if (Array.isArray(pathsOrTarget)) {
        return getSelectors(pathsOrTarget);
    } else {
        let path = [];
        while (pathsOrTarget) {
            path.push(pathsOrTarget);
            pathsOrTarget = pathsOrTarget.parentNode;
        }
        return getSelectors(path);
    }
}
```

#### 技术亮点
- **优先级处理**：ID > className > tagName
- **路径反转**：从子元素到父元素
- **过滤处理**：排除document和window对象
- **兼容性处理**：支持不同元素类型
- **递归遍历**：通过parentNode向上遍历DOM树
- **函数式编程**：使用map、filter、join等函数式方法

### 3. 交互事件捕获 (`getLastEvent.js`)

#### 功能概述
捕获用户最后的交互事件，为错误提供上下文信息。

#### 核心实现

**事件监听**
```javascript
let lastEvent;
['click', 'touchstart', 'mousedown', 'keydown', 'mouseover'].forEach(eventType => {
    document.addEventListener(eventType, (event) => {
            lastEvent = event;
            // html5之前使用mouseEvent.path等获取事件位置，
            // html5之后所有事件使用 PointerEven 替代，取消path
        }, {
            capture: true, // 捕获阶段执行 防止事件丢失
            passive: true, // 默认不阻止默认事件
        })
});

export default function () {
    return lastEvent;
}
```

#### 技术亮点
- **多事件监听**：覆盖主要用户交互事件
- **捕获阶段执行**：防止事件丢失
- **被动监听**：不阻止默认事件
- **HTML5兼容**：处理新旧标准差异
- **全局状态**：使用模块级变量保存最后事件

function blankScreen() {
    let wrapperElements = ["html", "body", "#container", ".content"];
    let emptyPoints = 0;

    function isWrapper(element) {
        let selector = getSelector(element);
        if (wrapperElements.indexOf(selector) !== -1) {
            emptyPoints++;
        }
    }

    onload(function () {
        // 在页面18个关键点进行采样
        for (let i = 0; i < 9; i++) {
            let xElements = document.elementsFromPoint(
                (window.innerWidth * i) / 10,
                window.innerHeight / 2
            );
            let yElements = document.elementsFromPoint(
                window.innerWidth / 2,
                (window.innerHeight * i) / 10
            );
            isWrapper(xElements[0]);
            isWrapper(yElements[0]);
        }
        
        // 如果空白点数量大于16个（共18个点），判定为白屏
        if (emptyPoints >= 16) {
            tracker.send({
                kind: "stability",
                type: "blank",
                emptyPoints: emptyPoints + "",
                screen: window.screen.width + "X" + window.screen.height,
                viewPoint: window.innerWidth + "X" + window.innerHeight,
                selector: getSelector(centerElements[0]),
            });
        }
    });
}
